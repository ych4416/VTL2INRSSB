;
;               VTLVTERM        : GET A VARIABLE TERM INCLUEDED INDEXED ONE.
;               ARG             : TOP OF STACK - STRING TO PARSE
;               RETURN          : TOP OF STACK - DESTINATION ADDRESS
;                                 2ND OF STACK - STRING TO NEXT PARSE

VTLVTERM        POP     VTLVTERMRA,STKPTR

                PUSH    VTLVTERMRA,STKPTR
                LINDR   VTLVTERMA0,VTLVTERMW0

                PUSH    VTLVTERMRA,STKPTR
                JMP     VTLVTERML99

                PUSH    VTLVTERMA0,STKPTR               ; ILLEGULAR VAR -> RETURN AS @
                MOVEM   VTLVARIABLESP,VTLVTERMW0
                ACCM    VTLVTERMW0,VTLVTERMNV
                INC     VTLVTERMW0
                PUSH    VTLVTERMW0,STKPTR
                PUSH    VTLVTERMRA,STKPTR
                JMP     VTLVTERML99

                PUSH    VTLVTERMW0,STKPTR
